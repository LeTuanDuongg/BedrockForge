cmake_minimum_required(VERSION 3.22)
project(BedrockForge VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_library(bedrockforge SHARED framework/core.cpp native-runtime/runtime.cpp sdk/scripting/native_bridge.cpp)
target_include_directories(bedrockforge PUBLIC sdk/include PRIVATE framework native-runtime)
target_compile_definitions(bedrockforge PRIVATE BF_BUILD)
if(MSVC)
  target_compile_options(bedrockforge PRIVATE /W4)
else()
  target_compile_options(bedrockforge PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(ANDROID)
  add_library(bedrockforge_host SHARED android-host/host_jni.cpp)
  target_include_directories(bedrockforge_host PRIVATE framework native-runtime)
  target_link_libraries(bedrockforge_host PRIVATE bedrockforge log dl)
endif()
option(BF_LEVI "Build the Android Levi lifecycle bridge" OFF)
if(BF_LEVI)
  if(NOT ANDROID)
    message(FATAL_ERROR "Levi bridge requires Android NDK")
  endif()
  if(NOT EXISTS "${PRELOADER_ANDROID_ROOT}/include/pl/Mod.hpp")
    message(FATAL_ERROR "Set PRELOADER_ANDROID_ROOT to audited preloader checkout")
  endif()
  execute_process(COMMAND git -C ${PRELOADER_ANDROID_ROOT} rev-parse HEAD
                  OUTPUT_VARIABLE BF_PRELOADER_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT BF_PRELOADER_COMMIT STREQUAL "5d108da350ce3e7f13f51415f81e65344ecd5d14")
    message(FATAL_ERROR "Preloader must match audited commit 5d108da350ce3e7f13f51415f81e65344ecd5d14")
  endif()
  add_subdirectory(${PRELOADER_ANDROID_ROOT} ${CMAKE_BINARY_DIR}/preloader)
  add_library(bedrockforge_levi SHARED android-host/levi_bridge.cpp)
  target_include_directories(bedrockforge_levi PRIVATE native-runtime framework)
  target_link_libraries(bedrockforge_levi PRIVATE bedrockforge preloader)
  set_target_properties(bedrockforge bedrockforge_levi PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
endif()
include(CTest)
if(BUILD_TESTING AND NOT ANDROID)
  add_executable(core_tests tests/unit/core_tests.cpp)
  target_include_directories(core_tests PRIVATE framework native-runtime)
  target_link_libraries(core_tests PRIVATE bedrockforge)
  add_test(NAME core COMMAND core_tests)
  add_library(bedrockforge_test_plugin SHARED tests/fixtures/native_plugin.cpp)
  target_include_directories(bedrockforge_test_plugin PRIVATE sdk/include)
  add_executable(loader_tests tests/integration/loader_tests.cpp)
  target_include_directories(loader_tests PRIVATE framework native-runtime)
  target_link_libraries(loader_tests PRIVATE bedrockforge ${CMAKE_DL_LIBS})
  add_test(NAME native_plugin_lifecycle COMMAND loader_tests $<TARGET_FILE:bedrockforge_test_plugin>)
  add_executable(c_header_test tests/unit/c_header.c)
  target_include_directories(c_header_test PRIVATE sdk/include)
  add_test(NAME c_abi_header COMMAND c_header_test)
endif()
