cmake_minimum_required(VERSION 3.22)
project(HybridMCPE LANGUAGES CXX)
option(HYBRID_BOOTSTRAP_PRELOADER "Include the Levi v1.5.24 Preloader registration adapter" OFF)
set(HYBRID_MODIFIED_PROFILE_DIR "" CACHE PATH "Generated modified-game fingerprint header directory")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(hybridmcpe SHARED native/runtime.cpp native/engine_bridge.cpp)
target_include_directories(hybridmcpe PUBLIC sdk PRIVATE native)
target_link_libraries(hybridmcpe PRIVATE ${CMAKE_DL_LIBS})
target_compile_options(hybridmcpe PRIVATE -Wall -Wextra -Werror)
if(ANDROID)
  target_link_libraries(hybridmcpe PRIVATE log)
  add_library(hybrid_diagnostic SHARED mods/diagnostic.cpp)
  target_include_directories(hybrid_diagnostic PRIVATE sdk)
  target_link_libraries(hybrid_diagnostic PRIVATE log)
  target_link_options(hybridmcpe PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/native/exports.map")
  set_property(TARGET hybridmcpe APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/native/exports.map")
  target_sources(hybridmcpe PRIVATE android/jni.cpp)
  if(HYBRID_MODIFIED_PROFILE_DIR)
    if(NOT EXISTS "${HYBRID_MODIFIED_PROFILE_DIR}/engine_profile.h")
      message(FATAL_ERROR "Prepare the modified engine profile before building")
    endif()
    target_sources(hybridmcpe PRIVATE bootstrap/modified_game/startup.cpp)
    target_include_directories(hybridmcpe PRIVATE "${HYBRID_MODIFIED_PROFILE_DIR}")
    target_compile_definitions(hybridmcpe PRIVATE HYBRID_MODIFIED_GAME=1)
    target_link_libraries(hybridmcpe PRIVATE log)
    add_library(hybrid_bootstrap_check SHARED tests/bootstrap_lifecycle_mod.cpp)
    target_include_directories(hybrid_bootstrap_check PRIVATE sdk)
    target_link_libraries(hybrid_bootstrap_check PRIVATE log)
  endif()
  if(HYBRID_BOOTSTRAP_PRELOADER)
    set_property(TARGET hybridmcpe PROPERTY CXX_STANDARD 20)
    target_sources(hybridmcpe PRIVATE bootstrap/preloader/adapter.cpp)
    target_link_libraries(hybridmcpe PRIVATE log)

    add_library(hybrid_preloader_test_module SHARED
      bootstrap/preloader/adapter.cpp
      tests/preloader_runtime_stub.cpp)
    target_include_directories(hybrid_preloader_test_module PRIVATE
      sdk bootstrap/preloader tests)
    target_link_libraries(hybrid_preloader_test_module PRIVATE log)
    target_compile_options(hybrid_preloader_test_module PRIVATE -Wall -Wextra -Werror)
    set_target_properties(hybrid_preloader_test_module PROPERTIES
      CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON
      OUTPUT_NAME hybridmcpe_preloader_test)

    add_executable(hybrid_preloader_tests tests/preloader_adapter_tests.cpp)
    target_include_directories(hybrid_preloader_tests PRIVATE
      bootstrap/preloader tests)
    target_link_libraries(hybrid_preloader_tests PRIVATE ${CMAKE_DL_LIBS})
    target_compile_options(hybrid_preloader_tests PRIVATE -Wall -Wextra -Werror)
    set_target_properties(hybrid_preloader_tests PROPERTIES
      CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)
    add_dependencies(hybrid_preloader_tests hybrid_preloader_test_module)
  endif()
endif()
add_library(iron_chest SHARED mods/iron_chest.cpp)
target_include_directories(iron_chest PRIVATE sdk)
add_library(hybrid_container_core STATIC native/iron_chest_container.cpp)
target_include_directories(hybrid_container_core PUBLIC native)
target_compile_options(hybrid_container_core PRIVATE -Wall -Wextra -Werror)
include(CTest)
if(BUILD_TESTING)
  add_executable(iron_chest_core_tests tests/iron_chest_core_tests.cpp)
  target_link_libraries(iron_chest_core_tests PRIVATE hybrid_container_core)
  target_compile_options(iron_chest_core_tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME iron_chest_core_tests COMMAND iron_chest_core_tests)
  add_executable(hybrid_self_tests tests/sha_tests.cpp)
  target_include_directories(hybrid_self_tests PRIVATE native)
  add_test(NAME hybrid_self_tests COMMAND hybrid_self_tests)
  add_executable(hybrid_native_tests tests/native_tests.cpp)
  target_include_directories(hybrid_native_tests PRIVATE sdk native)
  target_link_libraries(hybrid_native_tests PRIVATE hybridmcpe)
  add_test(NAME hybrid_native_tests COMMAND hybrid_native_tests)
  if(ANDROID)
    add_library(hybrid_runtime_test_mod SHARED tests/runtime_lifecycle_mod.cpp)
    target_include_directories(hybrid_runtime_test_mod PRIVATE sdk)
    target_compile_options(hybrid_runtime_test_mod PRIVATE -Wall -Wextra -Werror -fexceptions)
    add_executable(hybrid_runtime_lifecycle_tests
      native/runtime.cpp
      bootstrap/preloader/adapter.cpp
      tests/runtime_engine_stub.cpp
      tests/runtime_lifecycle_tests.cpp)
    target_include_directories(hybrid_runtime_lifecycle_tests PRIVATE sdk native bootstrap/preloader)
    target_link_libraries(hybrid_runtime_lifecycle_tests PRIVATE ${CMAKE_DL_LIBS} log)
    target_compile_options(hybrid_runtime_lifecycle_tests PRIVATE -Wall -Wextra -Werror -fexceptions)
    add_test(NAME hybrid_runtime_lifecycle_tests COMMAND hybrid_runtime_lifecycle_tests
      $<TARGET_FILE:hybrid_runtime_test_mod>)
  endif()
endif()
